````python path=src/gridbot/dual_utils.py mode=EDIT
"""
双账户通用工具
"""

import asyncio
from typing import List, Tuple, Any, Callable
from decimal import Decimal

class DualAccountUtils:
    """双账户通用工具类"""
    
    @staticmethod
    async def execute_on_both(
        long_obj: Any, 
        short_obj: Any, 
        method_name: str, 
        *args, **kwargs
    ) -> Tuple[Any, Any]:
        """在双账户对象上执行相同方法"""
        long_method = getattr(long_obj, method_name)
        short_method = getattr(short_obj, method_name)
        
        results = await asyncio.gather(
            long_method(*args, **kwargs),
            short_method(*args, **kwargs),
            return_exceptions=True
        )
        
        return results[0], results[1]
    
    @staticmethod
    async def parallel_cleanup(strategies: List[Any]) -> bool:
        """并行清理多个策略"""
        cleanup_tasks = []
        for strategy in strategies:
            if strategy and hasattr(strategy, '_cleanup_exchange_state'):
                cleanup_tasks.append(strategy._cleanup_exchange_state())
        
        if cleanup_tasks:
            results = await asyncio.gather(*cleanup_tasks, return_exceptions=True)
            # 检查是否有异常
            for result in results:
                if isinstance(result, Exception):
                    print(f"⚠️ 清理异常: {result}")
                    return False
        
        return True
    
    @staticmethod
    def format_account_stats(strategy: Any, side: str) -> str:
        """格式化账户统计信息"""
        try:
            if not strategy or not hasattr(strategy, 'grid_levels'):
                return f"{side}: 数据不可用"
            
            total_grids = len(strategy.grid_levels)
            pending_orders = sum(1 for level in strategy.grid_levels.values() 
                               if level.status == "ORDER_PENDING")
            held_positions = sum(1 for level in strategy.grid_levels.values() 
                               if level.status == "POSITION_HELD")
            total_profit = getattr(strategy, 'total_realized_profit', 0)
            
            return f"网格{total_grids} | 挂单{pending_orders} | 持仓{held_positions} | 利润{total_profit}"
            
        except Exception as e:
            return f"{side}: 统计异常 {e}"
    
    @staticmethod
    def validate_dual_config(config) -> List[str]:
        """验证双账户配置"""
        errors = []
        
        # 检查API密钥
        if not config.long_api_key or not config.long_api_secret:
            errors.append("多头账户API密钥不能为空")
        
        if not config.short_api_key or not config.short_api_secret:
            errors.append("空头账户API密钥不能为空")
        
        # 检查API密钥是否相同
        if (config.long_api_key == config.short_api_key and 
            config.long_api_secret == config.short_api_secret):
            errors.append("多头和空头账户不能使用相同的API密钥")
        
        # 检查价格区间
        if config.lower_price >= config.upper_price:
            errors.append("价格区间设置错误：下限价格必须小于上限价格")
        
        # 检查网格数量
        if config.grids < 2:
            errors.append("网格数量至少为2")
        
        return errors
````

## 专业评价

### 优点 ✅

1. **架构设计合理**
   - 保持了现有代码的完整性
   - 采用组合模式，而非继承，降低耦合
   - 职责分离清晰：配置、管理、执行分开

2. **兼容性良好**
   - 完全保留现有的挂单逻辑
   - 单账户和双账户功能独立
   - 不影响现有功能的使用

3. **错误处理完善**
   - 故障联动机制设计合理
   - 异常情况下的清理逻辑完整
   - 优雅退出机制

4. **可扩展性强**
   - 为后续动态网格功能预留接口
   - 模块化设计便于功能扩展

### 需要改进的地方 ⚠️

1. **代码重复**
   - 双账户操作存在大量重复代码
   - 缺少通用的双账户操作抽象

2. **配置验证分散**
   - 配置验证逻辑在多个地方重复
   - 缺少统一的配置验证入口

3. **错误处理可以更细化**
   - 某些异常处理过于宽泛
   - 缺少具体的错误分类和处理策略

4. **监控信息可以更丰富**
   - 当前监控信息相对简单
   - 缺少性能指标和详细统计

````python path=src/gridbot/dual_account_manager.py mode=EDIT
"""
双账户管理器 - 优化版
"""

import asyncio
from typing import Dict, Optional, Tuple
from decimal import Decimal
from .models import DualAccountConfig, BotConfig
from .exchange import ExchangeInterface
from .strategy import GridStrategy
from .dual_utils import DualAccountUtils

class DualAccountManager:
    """双账户管理器 - 优化版"""
    
    def __init__(self, config: DualAccountConfig):
        self.config = config
        self.utils = DualAccountUtils()
        
        # 验证配置
        config_errors = self.utils.validate_dual_config(config)
        if config_errors:
            raise ValueError(f"配置错误: {'; '.join(config_errors)}")
        
        # 创建双账户配置
        self.long_config = config.to_single_config("long")
        self.short_config = config.to_single_config("short")
        
        # 初始化状态
        self.running = False
        self.initialization_complete = False
        self._init_components()
    
    def _init_components(self):
        """初始化组件"""
        self.long_exchange: Optional[ExchangeInterface] = None
        self.short_exchange: Optional[ExchangeInterface] = None
        self.long_strategy: Optional[GridStrategy] = None
        self.short_strategy: Optional[GridStrategy] = None
        self.long_running = False
        self.short_running = False
    
    async def initialize(self, fresh_start: bool = False):
        """初始化双账户"""
        print(f"🔄 开始初始化双账户管理器...")
        
        try:
            await self._create_exchanges()
            await self._initialize_exchanges()
            await self._check_balances()
            await self._create_strategies()
            
            if fresh_start:
                await self._cleanup_both_accounts()
            
            await self._initialize_grids(fresh_start)
            
            self.initialization_complete = True
            print("🎉 双账户管理器初始化成功")
            
        except Exception as e:
            print(f"❌ 双账户初始化失败: {e}")
            await self._cleanup_on_error()
            raise
    
    async def _create_exchanges(self):
        """创建交易所接口"""
        self.long_exchange = ExchangeInterface(self.long_config)
        self.short_exchange = ExchangeInterface(self.short_config)
    
    async def _initialize_exchanges(self):
        """初始化交易所连接"""
        await self.utils.execute_on_both(
            self.long_exchange, self.short_exchange, 'initialize'
        )
        
        if self.config.market_type == 'future':
            await self.utils.execute_on_both(
                self.long_exchange, self.short_exchange, 
                'set_leverage_and_margin_mode'
            )
        
        print("✅ 双账户交易所连接初始化完成")
    
    async def _create_strategies(self):
        """创建策略实例"""
        self.long_strategy = GridStrategy(self.long_config, self.long_exchange)
        self.short_strategy = GridStrategy(self.short_config, self.short_exchange)
        print("✅ 双账户策略实例创建完成")
    
    async def _initialize_grids(self, fresh_start: bool):
        """初始化网格"""
        print("🚀 开始初始化双账户网格...")
        await self.utils.execute_on_both(
            self.long_strategy, self.short_strategy, 
            'initialize_grid', fresh_start
        )
        print("✅ 双账户网格初始化完成")
    
    async def _cleanup_both_accounts(self):
        """清理双账户状态"""
        print("🧹 开始清理双账户状态...")
        success = await self.utils.parallel_cleanup([
            self.long_strategy, self.short_strategy
        ])
        if success:
            print("✅ 双账户状态清理完成")
        else:
            print("⚠️ 双账户状态清理部分失败")
    
    def _get_account_stats(self, strategy: GridStrategy, side: str) -> str:
        """获取账户统计信息"""
        return self.utils.format_account_stats(strategy, side)
````
