class ProfitRecord(BaseModel):
    """利润记录模型"""
    grid_id: str
    open_price: Decimal
    close_price: Decimal
    amount: Decimal
    profit_usdt: Decimal
    profit_percentage: Decimal
    timestamp: int
    side: Literal["long", "short"]

class GridLevelState(BaseModel):
    id: str
    price: Decimal
    status: Literal["AVAILABLE", "ORDER_PENDING", "POSITION_HELD"]
    open_order_id: Optional[str] = None
    close_order_id: Optional[str] = None
    position_amount: Optional[Decimal] = None
    # 新增利润追踪字段
    open_timestamp: Optional[int] = None
    total_profit: Decimal = Decimal('0')
    trade_count: int = 0

class GridStrategy:
    def __init__(self, config: BotConfig, exchange: ExchangeInterface):
        # 现有初始化代码...
        self.profit_records: List[ProfitRecord] = []
        self.total_realized_profit = Decimal('0')
        self.profit_file_path = f"profit_records_{self.config.name.replace('/', '_')}.json"

    async def _handle_close_order_fill(self, level: GridLevelState, trade: Trade):
        """处理平仓订单成交的逻辑，完成一个完整周期并计算利润"""
        print(f"价格 {level.price} 的平仓订单成交，周期完成，计算利润...")

        # 计算本次交易利润
        profit_usdt = self._calculate_grid_profit(level, trade)
        
        # 记录利润
        profit_record = ProfitRecord(
            grid_id=level.id,
            open_price=level.price,
            close_price=trade.price,
            amount=trade.amount,
            profit_usdt=profit_usdt,
            profit_percentage=(profit_usdt / (level.price * trade.amount)) * 100,
            timestamp=trade.timestamp,
            side=self.side
        )
        
        self.profit_records.append(profit_record)
        self.total_realized_profit += profit_usdt
        
        # 更新网格层级统计
        level.total_profit += profit_usdt
        level.trade_count += 1
        
        print(f"✅ 网格 {level.id} 完成交易，利润: {profit_usdt:.4f} USDT ({profit_record.profit_percentage:.2f}%)")
        
        # 保存利润记录
        await self._save_profit_records()

        # 原有的重新放置开仓订单逻辑...
        level.status = "AVAILABLE"
        level.close_order_id = None
        level.position_amount = None
        level.open_timestamp = None

    def _calculate_grid_profit(self, level: GridLevelState, close_trade: Trade) -> Decimal:
        """计算单个网格的利润"""
        if self.side == "long":
            # 做多：买入价格低，卖出价格高，利润 = (卖出价 - 买入价) * 数量
            profit = (close_trade.price - level.price) * close_trade.amount
        else:
            # 做空：卖出价格高，买入价格低，利润 = (卖出价 - 买入价) * 数量
            profit = (level.price - close_trade.price) * close_trade.amount
        
        # 扣除手续费（假设手续费率0.1%）
        fee_rate = Decimal('0.001')
        total_fee = (level.price * close_trade.amount + close_trade.price * close_trade.amount) * fee_rate
        
        return profit - total_fee

    async def _save_profit_records(self):
        """保存利润记录到文件"""
        try:
            profit_data = {
                'total_realized_profit': str(self.total_realized_profit),
                'records': [json.loads(record.json()) for record in self.profit_records]
            }
            with open(self.profit_file_path, 'w') as f:
                json.dump(profit_data, f, indent=4)
        except Exception as e:
            print(f"保存利润记录时出错：{e}")

    async def _load_profit_records(self):
        """加载历史利润记录"""
        try:
            if os.path.exists(self.profit_file_path):
                with open(self.profit_file_path, 'r') as f:
                    data = json.load(f)
                    self.total_realized_profit = Decimal(data.get('total_realized_profit', '0'))
                    self.profit_records = [ProfitRecord(**record) for record in data.get('records', [])]
                    print(f"加载了 {len(self.profit_records)} 条利润记录，总利润: {self.total_realized_profit} USDT")
        except Exception as e:
            print(f"加载利润记录时出错：{e}")

# bot.py

def _calculate_total_profit(self) -> float:
    """计算已完成交易的总利润"""
    return float(self.strategy.total_realized_profit)

def _calculate_period_profit(self, hours: int) -> float:
    """计算指定时间段内的利润"""
    if not self.strategy.profit_records:
        return 0.0
    
    cutoff_time = int((datetime.now().timestamp() - hours * 3600) * 1000)
    period_profit = sum(
        record.profit_usdt for record in self.strategy.profit_records
        if record.timestamp >= cutoff_time
    )
    return float(period_profit)

def _calculate_unrealized_profit(self) -> float:
    """计算未实现利润（当前持仓的浮动盈亏）"""
    if not self.current_price:
        return 0.0
    
    unrealized_profit = Decimal('0')
    for level in self.strategy.grid_levels.values():
        if level.status == "POSITION_HELD" and level.position_amount:
            if self.strategy.side == "long":
                # 做多：当前价格 - 开仓价格
                pnl = (self.current_price - level.price) * level.position_amount
            else:
                # 做空：开仓价格 - 当前价格
                pnl = (level.price - self.current_price) * level.position_amount
            unrealized_profit += pnl
    
    return float(unrealized_profit)

def _update_stats(self, trade: Optional[Trade] = None):
    """更新并发送统计信息到前端"""
    self._print_strategy_status()

    if not self.config.frontend:
        return

    stats = {
        'total_profit': self._calculate_total_profit(),
        'unrealized_profit': self._calculate_unrealized_profit(),
        'daily_profit': self._calculate_period_profit(hours=24),
        'weekly_profit': self._calculate_period_profit(hours=168),
        'monthly_profit': self._calculate_period_profit(hours=720),
        'total_trades': len(self.strategy.profit_records),
        'win_rate': self._calculate_win_rate(),
    }

    if trade:
        self.ws_manager.add_price(trade.price)
        self.ws_manager.send_update('trade', trade.dict(), stats)
    else:
        self.ws_manager.send_update('stats', {}, stats)

def _calculate_win_rate(self) -> float:
    """计算胜率"""
    if not self.strategy.profit_records:
        return 0.0
    
    winning_trades = sum(1 for record in self.strategy.profit_records if record.profit_usdt > 0)
    return (winning_trades / len(self.strategy.profit_records)) * 100

# strategy.py

async def check_order_health(self):
    """定期检查订单健康状态并自动修复异常"""
    print("🔍 开始订单健康检查...")
    
    try:
        # 获取交易所当前所有挂单
        open_orders = await self.exchange.fetch_open_orders()
        open_order_ids = {order['id']: order for order in open_orders}
        
        # 获取当前持仓
        positions = await self.exchange.fetch_positions([self.config.pair])
        current_position = self._get_current_position(positions)
        
        health_issues = []
        
        # 检查各种健康问题
        await self._check_missing_orders(open_order_ids, health_issues)
        await self._check_orphaned_orders(open_order_ids, health_issues)
        await self._check_position_consistency(current_position, health_issues)
        await self._check_price_deviation(health_issues)
        
        # 修复发现的问题
        if health_issues:
            print(f"⚠️ 发现 {len(health_issues)} 个健康问题，开始修复...")
            await self._fix_health_issues(health_issues)
        else:
            print("✅ 订单健康检查通过，无异常")
            
    except Exception as e:
        print(f"❌ 订单健康检查时出错：{e}")

async def _check_missing_orders(self, open_order_ids: Dict[str, Any], health_issues: List[Dict]):
    """检查缺失的订单"""
    for grid_id, level in self.grid_levels.items():
        # 检查开仓订单
        if level.status == "ORDER_PENDING" and level.open_order_id:
            if level.open_order_id not in open_order_ids:
                health_issues.append({
                    'type': 'missing_open_order',
                    'grid_id': grid_id,
                    'level': level,
                    'description': f"网格 {grid_id} 的开仓订单 {level.open_order_id} 在交易所不存在"
                })
        
        # 检查平仓订单
        if level.status == "POSITION_HELD" and level.close_order_id:
            if level.close_order_id not in open_order_ids:
                health_issues.append({
                    'type': 'missing_close_order',
                    'grid_id': grid_id,
                    'level': level,
                    'description': f"网格 {grid_id} 的平仓订单 {level.close_order_id} 在交易所不存在"
                })

async def _check_orphaned_orders(self, open_order_ids: Dict[str, Any], health_issues: List[Dict]):
    """检查孤儿订单（交易所有但本地没有记录的订单）"""
    tracked_order_ids = set()
    
    # 收集所有被追踪的订单ID
    for level in self.grid_levels.values():
        if level.open_order_id:
            tracked_order_ids.add(level.open_order_id)
        if level.close_order_id:
            tracked_order_ids.add(level.close_order_id)
    
    # 查找孤儿订单
    for order_id, order in open_order_ids.items():
        if order['symbol'] == self.config.pair and order_id not in tracked_order_ids:
            health_issues.append({
                'type': 'orphaned_order',
                'order_id': order_id,
                'order': order,
                'description': f"发现孤儿订单 {order_id}，价格 {order['price']}"
            })

async def _check_position_consistency(self, current_position: Dict, health_issues: List[Dict]):
    """检查持仓一致性"""
    # 计算本地记录的持仓
    local_position = Decimal('0')
    position_count = 0
    
    for level in self.grid_levels.values():
        if level.status == "POSITION_HELD" and level.position_amount:
            if self.side == "long":
                local_position += level.position_amount
            else:
                local_position -= level.position_amount
            position_count += 1
    
    # 获取交易所实际持仓
    exchange_position = Decimal(str(current_position.get('contracts', '0')))
    
    # 允许小幅误差（0.1%）
    tolerance = abs(local_position) * Decimal('0.001')
    position_diff = abs(local_position - exchange_position)
    
    if position_diff > tolerance:
        health_issues.append({
            'type': 'position_mismatch',
            'local_position': local_position,
            'exchange_position': exchange_position,
            'difference': position_diff,
            'description': f"持仓不一致：本地 {local_position}，交易所 {exchange_position}"
        })
    
    # 检查持仓数量是否超限
    if position_count > self.config.max_position_count:
        health_issues.append({
            'type': 'position_overlimit',
            'current_count': position_count,
            'max_count': self.config.max_position_count,
            'description': f"持仓数量超限：当前 {position_count}，最大 {self.config.max_position_count}"
        })

async def _check_price_deviation(self, health_issues: List[Dict]):
    """检查价格偏离"""
    try:
        ticker = await self.exchange.fetch_ticker()
        current_price = Decimal(str(ticker['last']))
        
        # 检查是否偏离网格范围
        if current_price < self.lower_price or current_price > self.upper_price:
            deviation_pct = abs(current_price - (self.upper_price + self.lower_price) / 2) / current_price * 100
            health_issues.append({
                'type': 'price_deviation',
                'current_price': current_price,
                'deviation_percent': deviation_pct,
                'description': f"价格偏离网格范围：当前 {current_price}，范围 [{self.lower_price}, {self.upper_price}]"
            })
    except Exception as e:
        print(f"检查价格偏离时出错：{e}")

async def _fix_health_issues(self, health_issues: List[Dict]):
    """修复健康问题"""
    for issue in health_issues:
        try:
            if issue['type'] == 'missing_open_order':
                await self._fix_missing_open_order(issue)
            elif issue['type'] == 'missing_close_order':
                await self._fix_missing_close_order(issue)
            elif issue['type'] == 'orphaned_order':
                await self._fix_orphaned_order(issue)
            elif issue['type'] == 'position_mismatch':
                await self._fix_position_mismatch(issue)
            elif issue['type'] == 'position_overlimit':
                await self._fix_position_overlimit(issue)
            elif issue['type'] == 'price_deviation':
                await self._fix_price_deviation(issue)
                
        except Exception as e:
            print(f"修复问题 {issue['type']} 时出错：{e}")

async def _fix_missing_open_order(self, issue: Dict):
    """修复缺失的开仓订单"""
    level = issue['level']
    grid_id = issue['grid_id']
    
    # 检查订单是否已成交
    try:
        order = await self.exchange.fetch_order(level.open_order_id)
        if order['status'] == 'closed':
            print(f"🔧 修复：开仓订单 {level.open_order_id} 已成交，更新状态")
            trade_data = {
                'order_id': order['id'],
                'side': order['side'],
                'symbol': order['symbol'],
                'price': Decimal(str(order.get('average', order['price']))),
                'amount': Decimal(str(order.get('filled', order['amount']))),
                'cost': Decimal(str(order.get('cost', '0'))),
                'timestamp': int(order.get('timestamp', datetime.now().timestamp() * 1000))
            }
            trade = Trade(**trade_data)
            await self._handle_open_order_fill(level, trade)
        else:
            print(f"🔧 修复：重新放置开仓订单，网格 {grid_id}")
            level.status = "AVAILABLE"
            level.open_order_id = None
            await self._place_single_grid_order(level)
    except:
        # 订单不存在，重新放置
        print(f"🔧 修复：重新放置开仓订单，网格 {grid_id}")
        level.status = "AVAILABLE"
        level.open_order_id = None
        await self._place_single_grid_order(level)

async def _fix_missing_close_order(self, issue: Dict):
    """修复缺失的平仓订单"""
    level = issue['level']
    print(f"🔧 修复：重新放置平仓订单，网格 {issue['grid_id']}")
    await self._place_close_order_from_sync(level)

async def _fix_orphaned_order(self, issue: Dict):
    """修复孤儿订单"""
    order_id = issue['order_id']
    print(f"🔧 修复：取消孤儿订单 {order_id}")
    try:
        await self.exchange.cancel_order(order_id)
        print(f"✅ 已取消孤儿订单 {order_id}")
    except Exception as e:
        print(f"❌ 取消孤儿订单 {order_id} 失败：{e}")

def _get_current_position(self, positions: List[Dict]) -> Dict:
    """获取当前持仓信息"""
    for position in positions:
        if position.get('symbol') == self.config.pair:
            side = position.get('side')
            if (self.side == 'long' and side == 'long') or (self.side == 'short' and side == 'short'):
                return position
    return {}

async def _place_single_grid_order(self, level: GridLevelState):
    """为单个网格放置开仓订单"""
    try:
        order_amount_coin = self.config.order_amount_usdt / level.price
        params = {'positionSide': 'LONG' if self.side == 'long' else 'SHORT'}

        if self.side == "long":
            order = await self.exchange.create_limit_buy_order(order_amount_coin, level.price, params)
        else:
            order = await self.exchange.create_limit_sell_order(order_amount_coin, level.price, params)

        level.open_order_id = order['id']
        level.status = "ORDER_PENDING"
        print(f"✅ 重新放置开仓订单：网格 {level.id}，订单 {order['id']}")
        
    except Exception as e:
        print(f"❌ 重新放置开仓订单失败，网格 {level.id}：{e}")
